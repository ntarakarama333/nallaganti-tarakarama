analytics_pipeline.sql 
SELECT department, 
       COUNT(employee_id) AS total_staff,
       ROUND(AVG(performance_score), 2) AS avg_kpi
FROM enterprise_data
WHERE status = 'Active'
GROUP BY department
HAVING avg_kpi > 8.5
ORDER BY avg_kpi DESC;
